cmake_minimum_required(VERSION 3.18)
project(unified-cache-management VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(UCM_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR})

option(BUILD_UCM_STORE "build ucm store module." ON)
option(BUILD_UCM_SPARSE "build ucm sparse module." OFF)
option(BUILD_UCM_MINDIE "build ucm MindIE integration module." OFF)
option(BUILD_UNIT_TESTS "build all unit test suits." OFF)
option(BUILD_NUMA "build numactl library." OFF)
option(UCM_ENABLE_GDR_STREAM "build GPUDirect RDMA stream support for CUDA runtime." OFF)
option(ASCEND_SUPPORTS_REGISTER_PIN "enable Ascend register pin optimization (requires CANN >= 8.5)" OFF)
option(DOWNLOAD_DEPENDENCE "download dependence by cmake." ON)
option(UCM_RELEASE_BUILD "build a source-bound native release wheel" OFF)
set(RUNTIME_ENVIRONMENT "simu" CACHE STRING "runtime: simu, ascend, ascend-a3, musa or cuda.")
set(UCM_CXX11_ABI "" CACHE STRING "glibc++ ABI for MindIE builds: 0 or 1")
set(UCM_RELEASE_PROFILE "" CACHE STRING "release wheel profile")
set(UCM_RELEASE_SPEC_ID "" CACHE STRING "release wheel spec")
set(UCM_RELEASE_SOURCE_SHA "" CACHE STRING "release source commit")
set(UCM_RELEASE_VERSION "" CACHE STRING "controlled local wheel version")
set(UCM_RELEASE_BUILD_KEY "" CACHE STRING "canonical release task digest")
set(UCM_RELEASE_REQUIRED_TARGETS "" CACHE STRING "exact required native targets")
set(UCM_RELEASE_FORBIDDEN_TARGETS "" CACHE STRING "exact forbidden native targets")
set(SOURCE_DATE_EPOCH "" CACHE STRING "reproducible build timestamp")

if(UCM_RELEASE_BUILD)
    foreach(REQUIRED_VALUE
            UCM_RELEASE_PROFILE UCM_RELEASE_SPEC_ID UCM_RELEASE_SOURCE_SHA
            UCM_RELEASE_VERSION UCM_RELEASE_BUILD_KEY UCM_RELEASE_REQUIRED_TARGETS
            UCM_RELEASE_FORBIDDEN_TARGETS SOURCE_DATE_EPOCH)
        if("${${REQUIRED_VALUE}}" STREQUAL "")
            message(FATAL_ERROR "${REQUIRED_VALUE} is required for UCM_RELEASE_BUILD")
        endif()
    endforeach()
    string(LENGTH "${UCM_RELEASE_SOURCE_SHA}" UCM_RELEASE_SOURCE_SHA_LENGTH)
    if(NOT UCM_RELEASE_SOURCE_SHA_LENGTH EQUAL 40 OR NOT UCM_RELEASE_SOURCE_SHA MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "UCM_RELEASE_SOURCE_SHA must be a full lowercase commit")
    endif()
    string(LENGTH "${UCM_RELEASE_BUILD_KEY}" UCM_RELEASE_BUILD_KEY_LENGTH)
    if(NOT UCM_RELEASE_BUILD_KEY_LENGTH EQUAL 71 OR NOT UCM_RELEASE_BUILD_KEY MATCHES "^sha256:[0-9a-f]+$")
        message(FATAL_ERROR "UCM_RELEASE_BUILD_KEY must be an immutable SHA256")
    endif()
    if(NOT SOURCE_DATE_EPOCH MATCHES "^[0-9]+$")
        message(FATAL_ERROR "SOURCE_DATE_EPOCH must be an integer")
    endif()
    if(BUILD_UCM_MINDIE OR BUILD_UCM_SPARSE)
        message(FATAL_ERROR "MindIE and sparse targets are forbidden in release wheels")
    endif()
    set(BUILD_UCM_MINDIE OFF CACHE BOOL "" FORCE)
    set(BUILD_UCM_SPARSE OFF CACHE BOOL "" FORCE)
    set(UCM_RELEASE_DS3FS_POLICY "FORBIDDEN" CACHE STRING "" FORCE)
    if(RUNTIME_ENVIRONMENT STREQUAL "cuda")
        set(UCM_RELEASE_MOONCAKE_POLICY "FORBIDDEN" CACHE STRING "" FORCE)
    elseif(RUNTIME_ENVIRONMENT STREQUAL "ascend" OR RUNTIME_ENVIRONMENT STREQUAL "ascend-a3")
        set(UCM_RELEASE_MOONCAKE_POLICY "REQUIRED" CACHE STRING "" FORCE)
    else()
        message(FATAL_ERROR "release RUNTIME_ENVIRONMENT cannot fall back to ${RUNTIME_ENVIRONMENT}")
    endif()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" UCM_NATIVE_PROCESSOR)
    if(UCM_NATIVE_PROCESSOR MATCHES "^(x86_64|amd64)$")
        set(UCM_NATIVE_ARCH "amd64")
        set(UCM_ASCEND_ARCH_SUBDIR "x86_64-linux")
    elseif(UCM_NATIVE_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(UCM_NATIVE_ARCH "arm64")
        set(UCM_ASCEND_ARCH_SUBDIR "aarch64-linux")
    else()
        message(FATAL_ERROR "unsupported native release architecture: ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    if(NOT UCM_RELEASE_SPEC_ID STREQUAL "${UCM_RELEASE_PROFILE}-${UCM_NATIVE_ARCH}")
        message(FATAL_ERROR "release spec architecture does not match the native builder")
    endif()
    if(RUNTIME_ENVIRONMENT STREQUAL "ascend-a3")
        set(ASCEND_ROOT "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "" FORCE)
        set(ASCEND_ARCH_DIR "${ASCEND_ROOT}/${UCM_ASCEND_ARCH_SUBDIR}" CACHE PATH "" FORCE)
    endif()
endif()

set(UCM_RUNTIME_ASCEND_FAMILY OFF)
set(UCM_RUNTIME_ASCEND_SDMA_DIRECT OFF)
if(RUNTIME_ENVIRONMENT STREQUAL "ascend")
    set(UCM_RUNTIME_ASCEND_FAMILY ON)
elseif(RUNTIME_ENVIRONMENT STREQUAL "ascend-a3")
    set(UCM_RUNTIME_ASCEND_FAMILY ON)
    set(UCM_RUNTIME_ASCEND_SDMA_DIRECT ON)
endif()

if(UCM_RELEASE_BUILD)
    set(UCM_COMMIT_ID "${UCM_RELEASE_SOURCE_SHA}")
else()
    execute_process(COMMAND git rev-parse HEAD OUTPUT_VARIABLE UCM_COMMIT_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
endif()
add_compile_definitions(UCM_PROJECT_NAME="${PROJECT_NAME}")
add_compile_definitions(UCM_PROJECT_VERSION="${PROJECT_VERSION}")
add_compile_definitions(UCM_COMMIT_ID="${UCM_COMMIT_ID}")
add_compile_definitions(UCM_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
if(BUILD_UCM_MINDIE)
    if(NOT UCM_CXX11_ABI MATCHES "^[01]$")
        message(FATAL_ERROR "BUILD_UCM_MINDIE=ON requires -DUCM_CXX11_ABI=0 or 1")
    endif()
    add_compile_definitions(_GLIBCXX_USE_CXX11_ABI=${UCM_CXX11_ABI})
endif()

string(TOLOWER "${CMAKE_BUILD_TYPE}" CMAKE_BUILD_TYPE_LOWER)
set(FLAGS_PUBLIC "-Wall -Werror -fPIC -Wl,-z,relro,-z,now")
if(CMAKE_BUILD_TYPE_LOWER STREQUAL "release" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # GCC can report a false positive in the release build's vendored fmt v11.
    set(FLAGS_PUBLIC "${FLAGS_PUBLIC} -Wno-error=stringop-overflow")
endif()
set(FLAGS_DEBUG "-O0 -g")
set(FLAGS_RELEASE "-s -O2 -fstack-protector-strong -D_FORTIFY_SOURCE=2")
if(CMAKE_BUILD_TYPE_LOWER STREQUAL "debug")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${FLAGS_PUBLIC} ${FLAGS_DEBUG}")
else()
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${FLAGS_PUBLIC} ${FLAGS_RELEASE}")
endif()
if(BUILD_UNIT_TESTS)
    enable_testing()
endif()

add_subdirectory(ucm)
if(UCM_RELEASE_BUILD)
    foreach(REQUIRED_TARGET IN LISTS UCM_RELEASE_REQUIRED_TARGETS)
        if(NOT TARGET "${REQUIRED_TARGET}")
            message(FATAL_ERROR "required release target is absent: ${REQUIRED_TARGET}")
        endif()
    endforeach()
    foreach(FORBIDDEN_TARGET IN LISTS UCM_RELEASE_FORBIDDEN_TARGETS)
        if(TARGET "${FORBIDDEN_TARGET}")
            message(FATAL_ERROR "forbidden release target is present: ${FORBIDDEN_TARGET}")
        endif()
    endforeach()

    function(ucm_json_array OUTPUT_NAME LIST_NAME)
        set(JSON_VALUE "[")
        set(SEPARATOR "")
        foreach(ITEM IN LISTS ${LIST_NAME})
            string(APPEND JSON_VALUE "${SEPARATOR}\"${ITEM}\"")
            set(SEPARATOR ",")
        endforeach()
        string(APPEND JSON_VALUE "]")
        set(${OUTPUT_NAME} "${JSON_VALUE}" PARENT_SCOPE)
    endfunction()
    ucm_json_array(UCM_REQUIRED_JSON UCM_RELEASE_REQUIRED_TARGETS)
    ucm_json_array(UCM_FORBIDDEN_JSON UCM_RELEASE_FORBIDDEN_TARGETS)
    set(UCM_COMPONENT_MANIFEST "${CMAKE_BINARY_DIR}/ucm-native-components.json")
    file(WRITE "${UCM_COMPONENT_MANIFEST}"
        "{\"build_key\":\"${UCM_RELEASE_BUILD_KEY}\","
        "\"cpu_arch\":\"${UCM_NATIVE_ARCH}\","
        "\"forbidden_native\":${UCM_FORBIDDEN_JSON},"
        "\"installed_targets\":${UCM_REQUIRED_JSON},"
        "\"kind\":\"ucm-native-components\","
        "\"profile_id\":\"${UCM_RELEASE_PROFILE}\","
        "\"required_native\":${UCM_REQUIRED_JSON},"
        "\"schema_version\":1,"
        "\"source_sha\":\"${UCM_RELEASE_SOURCE_SHA}\","
        "\"spec_id\":\"${UCM_RELEASE_SPEC_ID}\","
        "\"version\":\"${UCM_RELEASE_VERSION}\"}\n")
    install(FILES "${UCM_COMPONENT_MANIFEST}" DESTINATION ucm COMPONENT ucm)
endif()
if(BUILD_UNIT_TESTS)
    add_subdirectory(test)
endif()
